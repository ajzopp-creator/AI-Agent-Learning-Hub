from shared_resources.python_utils.vault_interface import write_to_vault

    "signal_source": "P_115",

        "atm_at_signal": 1.85,
        "close_at_signal": 47.75,
        "trailing_volume_30d": 1850000,
        "signal_rationale": "Dip into 20-day MA after earnings"
    },
    "signal_metadata": {
        "p115_session_date": "2026-06-03",
        "p115_chart_timeframe": "1D",

    schema_name="P400SIG",
    data=signal_packet,
    overwrite=True
